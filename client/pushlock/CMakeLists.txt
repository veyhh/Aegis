cmake_minimum_required(VERSION 3.20)
project(aegis_pushlock_client LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "Build this client with a Windows compiler. Linux runs the Wine harness.")
endif()
add_executable(aegis_pushlock_client main.c)
target_include_directories(aegis_pushlock_client PRIVATE ../../driver/pushlock)
set_target_properties(aegis_pushlock_client PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
  set_property(TARGET aegis_pushlock_client PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(aegis_pushlock_client PRIVATE /W4 /WX)
endif()
